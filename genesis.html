<script>

    let itemsFoundCount = 4; 

    let currentScore = 3450;  

    

    const foundRegistry = {

        'noahs-ark': false, 'abrahams-tent': false, 'josephs-coat': false,

        'egyptian-grain': false, 'garden-tree': false, 'serpent': false,

        'apple': false, 'altar-of-stones': false, 'rainbow': false,

        'dove': true, 'well-of-water': true, 'camel': false,

        'golden-ring': false, 'pillar-of-salt': false, 'bundle-of-myrrh': false

    };


    const gameContainer = document.getElementById('game-container');

    const itemsCounterEl = document.getElementById('items-counter');

    const scoreDisplayEl = document.getElementById('score-display');


    // Master click logic handler

    gameContainer.addEventListener('click', function(e) {

        const target = e.target;

        

        if (target.classList.contains('hotspot')) {

            const itemKey = target.getAttribute('data-item');

            if (foundRegistry[itemKey]) return;

            

            foundRegistry[itemKey] = true;

            itemsFoundCount++;

            currentScore += 100;

            

            // Updates HUD interfaces instantly with formatted values

            itemsCounterEl.innerText = `ITEMS: ${itemsFoundCount} / 15`;

            scoreDisplayEl.innerText = currentScore.toLocaleString();

            

            createScorePopup(e.clientX, e.clientY);

            target.style.pointerEvents = 'none';

            

            if (itemsFoundCount === 15) {

                setTimeout(() => { alert("Glory! You have discovered all the foundational elements of Genesis!"); }, 500);

            }

        }

    });


    function createScorePopup(clientX, clientY) {

        const rect = gameContainer.getBoundingClientRect();

        const popup = document.createElement('div');

        popup.className = 'score-popup';

        popup.innerText = '+100 PTS';

        

        const xPercent = ((clientX - rect.left) / rect.width) * 100;

        const yPercent = ((clientY - rect.top) / rect.height) * 100;

        popup.style.left = `${xPercent}%`;

        popup.style.top = `${yPercent}%`;

        

        gameContainer.appendChild(popup);

        setTimeout(() => { popup.remove(); }, 1200);

    }


    document.getElementById('btn-replay').addEventListener('click', (e) => { e.stopPropagation(); alert('Restarting chapter layout...'); });

    document.getElementById('btn-pause').addEventListener('click', (e) => { e.stopPropagation(); alert('Game Paused.'); });

</script>
